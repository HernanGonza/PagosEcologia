    public function handlePaymentWebhook(Request $request)
    {
        Log::info('Webhook: '. time() .' - ' . $request->getContent());

        $json_data = $request->getContent();
        $array_data = json_decode($json_data, true);

        $payment = Payment::where('id_transaction', $array_data['TransaccionComercioId'])
            ->update([
                'state' => $array_data['Estado'],
                'type_transaction' => $array_data['Tipo'],
                'json_data' => $json_data,
            ]);

        if($payment == null){
            Transaction::create([
                'id_transaction' => $array_data['TransaccionComercioId'],
                'state' => $array_data['Estado'],
                'type_transaction' => $array_data['Tipo'],
                'json_data' => $json_data,
            ]);
        }

        $id_licence = explode('_', $array_data['TransaccionComercioId'])[0];
        $licencia = Licence::with('user')->where('id', $id_licence)->first();

        if($array_data['Estado'] == 'REALIZADA'){

            $user = User::where('email', $licencia->user->email)->first();
            $pdfPath = $this->savePdf($array_data['TransaccionComercioId'], $licencia, $user);
            $relativePath = Str::after($pdfPath, 'storage/app/public/');

            Licence::where('id', $id_licence)->update([
                'active' => true,
                'pdf_path' => $relativePath,
            ]);

            $user->notify(new LicenseEmail($pdfPath));
        }

        return response()->json(['message' => 'OK']);
    }

class PayLicence extends Page implements HasForms {

    use InteractsWithFormActions;

    public function mount($record) {
        $this->licence_id = $record;
        $this->preparePayment();
    }

    private function preparePayment() {
        $arr = GeneralList::where('code',
            Parameter::where('const', Parameters::PAYMENT->const())->value('code')
        )->pluck('s_value', 'value')->toArray();

        $json_data = Licence::where('id', $this->licence_id)->first()->json_data;
        $licence_data = json_decode($json_data, true);
        $this->montoBruto = (isset($licence_data['monto_bruto']))? $licence_data['monto_bruto'] : 0;
        $this->montoDescuento = (isset($licence_data['monto_descuento']))? $licence_data['monto_descuento'] : 0;

        $monto_cents = $licence_data['monto_licencia_cents'];
        $monto = $monto_cents / 100;

        $descripcion_licencia = $licence_data['category'];
        $this->url_sandbox = $arr['URL'];
        $this->secretKey = $arr['SECRETKEY'];
        $comercioId = $arr['COMERCIO'];

        if (empty($this->secretKey) || empty($comercioId)) {
            Log::error('Configuración de MacroClick incompleta');
            throw new \Exception('Configuración incompleta: Claves requeridas faltantes.');
        }

        $transaccionId = $this->licence_id.'_Time:'.time();
        $url = env('APP_URL');
        $callbackSuccess = $url.$arr['CALLBACK_SUCCESS'].'&licencia='.$this->licence_id;
        $callbackCancel = $url.$arr['CALLBACK_CANCEL'].'&licencia='.$this->licence_id;


        try {
            $aes = new AESEncrypter();
            $this->CallbackSuccess = $aes->EncryptString($callbackSuccess, $this->secretKey);
            $this->CallbackCancel = $aes->EncryptString($callbackCancel, $this->secretKey);
            $this->Comercio = $comercioId; // Sin encriptar
            $this->TransaccionComercioId = $transaccionId; // Sin encriptar
            $this->Monto = $aes->EncryptString($monto_cents, $this->secretKey);
            $this->Producto = $descripcion_licencia;// Sin encriptar
            $this->MontoProducto = number_format($monto, '2', '.');// Sin encriptar
            $this->InformacionAdicional = $aes->EncryptString($this->licence_id. ' Info Adicional', $this->secretKey);
            $this->SucursalComercio = $aes->EncryptString('0000000000', $this->secretKey);
            $this->hash = $this->generateSignature($monto_cents);

            $payment = Payment::where('id_licence', $this->licence_id)
                            ->whereNotIn('state', ['CANCELADO'])
                            ->where('id_user', Auth::id())
                            ->first();

            if($payment == null){
                Payment::create([
                    'id_transaction' => $transaccionId,
                    'id_user' => Auth::id(),
                    'id_licence'=> $this->licence_id,
                    'state' => Payments::CREADA->label(),
                    'type_transaction' => null,
                    'json_data' => null,
                ]);
            }
            else{
                Payment::updated([
                    'id_transaction' => $transaccionId,
                    'state' => Payments::CREADA->label(),
                    'type_transaction' => null,
                    'json_data' => null,
                ]);
            }

        } catch (\Exception $e) {
            Log::error('Error al encriptar datos para el pago: ' . $e->getMessage(), [
                'trace' => $e->getTraceAsString(),
            ]);
            throw new \Exception('Error al preparar los datos de pago. Consulte al administrador.'.$e->getMessage());
        }
    }

    protected function generateSignature($amount)
    {
        try {
            $SHA256Encript = new SHA256Encript();
            return $SHA256Encript->Generate(
                request()->ip(),
                $this->secretKey,
                $this->Comercio,
                $this->SucursalComercio,
                number_format($amount * 100, 0, '', '')
            );
        } catch (\Exception $e) {
            Log::error('Error generando firma digital: ' . $e->getMessage());
            throw new \Exception('Error al generar la firma digital.');
        }
    }

    if (!file_exists($path)) {
        abort(404, 'Archivo no encontrado.');
    }

    return response()->download($path, "licencia_pesca_{$paymentId}.pdf");
})->name('download.license');

Route::get('/storage-link', function(){
    $targetFolder = storage_path('app/public');
    $linkFolder = $_SERVER['DOCUMENT_ROOT'].'/storage';

    try {
        symlink($targetFolder,$linkFolder);
        echo 'Symlink process successfully completed';
    } catch (Exception $e) {
        Artisan::call('storage:link');
        echo 'Symlink Artisan process successfully completed';
    }
});

===== app/Models/Payment.php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Support\Facades\Auth;

class Payment extends Model {
    use HasFactory;

    protected $table = 'payments';

    protected $fillable = [
        'id_transaction',
        'id_user',
        'id_licence',
        'state',
        'type_transaction',
        'json_data',
        'created_by',
        'updated_by'
    ];

    protected $casts = [
        'json_data' => 'array'
    ];

    public function user() {
        return $this->belongsTo(User::class, 'id_user');
    }

    public function licence() {
        return $this->belongsTo(Licence::class, 'id_licence');
    }

    public function country() {
        return $this->belongsTo(Country::class);
    }

    public function municipios()
    {
        return $this->hasMany(City::class);
    }

    public function creator(): BelongsTo
    {
        return $this->belongsTo(User::class, 'created_by');
    }

    public function updater(): BelongsTo
    {
        return $this->belongsTo(User::class, 'updated_by');
    }

    protected static function booted()
    {
        static::creating(function ($model) {
            if (Auth::check()) {
                $model->created_by = Auth::id();
                $model->updated_at = null;
            }
        });

        static::updating(function ($model) {
            if (Auth::check()) {
                $model->updated_by = Auth::id();
            }
        });
    }
}
